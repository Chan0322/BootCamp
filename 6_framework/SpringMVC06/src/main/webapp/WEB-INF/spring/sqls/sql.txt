CREATE TABLE TESTBOARD(
	TESTNO NUMBER PRIMARY KEY,
	TESTNAME VARCHAR2(500) NOT NULL,
	TESTTITLE VARCHAR2(1000) NOT NULL,
	TESTCONTENT VARCHAR2(2000) NOT NULL,
	TESTDATE DATE NOT NULL
);
